cmake_minimum_required(VERSION 3.22.1)
project(gallery_gguf LANGUAGES C CXX)
option(GALLERY_GGUF_HOST_CHECK "Build the command-line inference check" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_LLGUIDANCE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
# Baseline ARM64, rather than instructions available only on the build host.
if(ANDROID)
  set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()

get_filename_component(GALLERY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../" ABSOLUTE)
set(LLAMA_SOURCE "${GALLERY_ROOT}/third_party/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Run git submodule update --init --recursive before building")
endif()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama")

add_library(gallery_gguf_core STATIC gguf_engine.cpp)
target_include_directories(gallery_gguf_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gallery_gguf_core PUBLIC llama llama-common)

if(ANDROID AND NOT GALLERY_GGUF_HOST_CHECK)
  add_library(gallery_gguf SHARED gguf_jni.cpp)
  target_link_libraries(gallery_gguf PRIVATE gallery_gguf_core log android)
  target_link_options(gallery_gguf PRIVATE "-Wl,-z,max-page-size=16384")
else()
  # The same engine can be checked on a host without an Android SDK.
  add_executable(gallery-gguf-smoke gguf_smoke.cpp)
  target_link_libraries(gallery-gguf-smoke PRIVATE gallery_gguf_core)
endif()
